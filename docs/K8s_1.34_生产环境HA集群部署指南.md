## Phase 1 · 所有节点基础准备（11 台）

### 1.1 设置主机名和 hosts

```bash
# 各节点设置对应主机名
sudo hostnamectl set-hostname <主机名>

# 所有节点统一 hosts
cat <<EOF | sudo tee -a /etc/hosts
10.0.0.5   lb-01
10.0.0.6   lb-02
10.0.0.11  etcd-01
10.0.0.12  etcd-02
10.0.0.13  etcd-03
10.0.0.21  k8s-master-01
10.0.0.22  k8s-master-02
10.0.0.23  k8s-master-03
10.0.0.31  k8s-gpu-01
10.0.0.32  k8s-gpu-02
10.0.0.100 k8s-api-vip
EOF
```

### 1.2 关闭 Swap（所有节点）

```bash
sudo swapoff -a
sudo sed -i '/ swap / s/^\(.*\)$/#\1/g' /etc/fstab
```

### 1.3 加载内核模块和参数（所有节点）

```bash
cat <<EOF | sudo tee /etc/modules-load.d/k8s.conf
overlay
br_netfilter
EOF
sudo modprobe overlay
sudo modprobe br_netfilter

cat <<EOF | sudo tee /etc/sysctl.d/k8s.conf
net.bridge.bridge-nf-call-iptables  = 1
net.bridge.bridge-nf-call-ip6tables = 1
net.ipv4.ip_forward                 = 1
EOF
sudo sysctl --system
```

### 1.4 关闭防火墙 + 时间同步（所有节点）

```bash
sudo systemctl disable --now ufw
sudo apt-get install -y chrony
sudo systemctl enable --now chrony
```

## Phase 2 · 生成 TLS 证书（在 etcd-01 上操作）

### 2.1 安装 cfssl

```bash
sudo apt-get install -y golang-cfssl
```

### 2.2 生成 CA

```bash
mkdir -p ~/etcd-certs && cd ~/etcd-certs

cat > ca-config.json <<EOF
{
  "signing": {
    "default": { "expiry": "87600h" },
    "profiles": {
      "etcd": {
        "usages": ["signing", "key encipherment", "server auth", "client auth"],
        "expiry": "87600h"
      }
    }
  }
}
EOF

cat > ca-csr.json <<EOF
{
  "CN": "etcd-ca",
  "key": { "algo": "rsa", "size": 2048 },
  "names": [{ "O": "etcd", "OU": "CA" }]
}
EOF

cfssl gencert -initca ca-csr.json | cfssljson -bare ca
```

### 2.3 生成 etcd 证书

```bash
cat > etcd-csr.json <<EOF
{
  "CN": "etcd",
  "hosts": [
    "127.0.0.1",
    "10.0.0.11", "10.0.0.12", "10.0.0.13",
    "etcd-01", "etcd-02", "etcd-03"
  ],
  "key": { "algo": "rsa", "size": 2048 },
  "names": [{ "O": "etcd", "OU": "etcd cluster" }]
}
EOF

cfssl gencert -ca=ca.pem -ca-key=ca-key.pem -config=ca-config.json -profile=etcd etcd-csr.json | cfssljson -bare etcd
```

### 2.4 分发证书

```bash
# 到 3 台 etcd
for node in etcd-01 etcd-02 etcd-03; do
  ssh $node "sudo mkdir -p /etc/etcd/ssl"
  scp ca.pem etcd.pem etcd-key.pem $node:/tmp/
  ssh $node "sudo mv /tmp/{ca.pem,etcd.pem,etcd-key.pem} /etc/etcd/ssl/ && sudo chmod 600 /etc/etcd/ssl/*-key.pem"
done

# 到 3 台 Master（API Server 连 etcd 用）
for node in k8s-master-01 k8s-master-02 k8s-master-03; do
  ssh $node "sudo mkdir -p /etc/kubernetes/pki/etcd"
  scp ca.pem $node:/tmp/etcd-ca.pem
  scp etcd.pem $node:/tmp/apiserver-etcd-client.pem
  scp etcd-key.pem $node:/tmp/apiserver-etcd-client-key.pem
  ssh $node "sudo mv /tmp/etcd-ca.pem /etc/kubernetes/pki/etcd/ca.crt && \
             sudo mv /tmp/apiserver-etcd-client.pem /etc/kubernetes/pki/apiserver-etcd-client.crt && \
             sudo mv /tmp/apiserver-etcd-client-key.pem /etc/kubernetes/pki/apiserver-etcd-client.key"
done
```

## Phase 3 · 部署 etcd 集群（3 台 etcd 节点）

### 3.1 下载 etcd

```bash
ETCD_VER=v3.5.17
curl -sL https://github.com/etcd-io/etcd/releases/download/${ETCD_VER}/etcd-${ETCD_VER}-linux-amd64.tar.gz | tar xz
sudo mv etcd-${ETCD_VER}-linux-amd64/etcd* /usr/local/bin/
etcd --version
```

### 3.2 创建 systemd 服务

每台 etcd 节点执行，**替换 THIS_NAME 和 THIS_IP**：

```bash
THIS_NAME=etcd-01        # etcd-01 / etcd-02 / etcd-03
THIS_IP=10.0.0.11        # .11 / .12 / .13

sudo mkdir -p /var/lib/etcd

cat <<EOF | sudo tee /etc/systemd/system/etcd.service
[Unit]
Description=etcd
After=network.target

[Service]
Type=notify
ExecStart=/usr/local/bin/etcd \\
  --name=${THIS_NAME} \\
  --data-dir=/var/lib/etcd \\
  --listen-client-urls=https://${THIS_IP}:2379,https://127.0.0.1:2379 \\
  --advertise-client-urls=https://${THIS_IP}:2379 \\
  --listen-peer-urls=https://${THIS_IP}:2380 \\
  --initial-advertise-peer-urls=https://${THIS_IP}:2380 \\
  --initial-cluster=etcd-01=https://10.0.0.11:2380,etcd-02=https://10.0.0.12:2380,etcd-03=https://10.0.0.13:2380 \\
  --initial-cluster-token=etcd-cluster-prod \\
  --initial-cluster-state=new \\
  --client-cert-auth=true \\
  --trusted-ca-file=/etc/etcd/ssl/ca.pem \\
  --cert-file=/etc/etcd/ssl/etcd.pem \\
  --key-file=/etc/etcd/ssl/etcd-key.pem \\
  --peer-client-cert-auth=true \\
  --peer-trusted-ca-file=/etc/etcd/ssl/ca.pem \\
  --peer-cert-file=/etc/etcd/ssl/etcd.pem \\
  --peer-key-file=/etc/etcd/ssl/etcd-key.pem
Restart=always
RestartSec=5
LimitNOFILE=65536

[Install]
WantedBy=multi-user.target
EOF
```

### 3.3 启动并验证

```bash
# 3 台同时启动
sudo systemctl daemon-reload
sudo systemctl enable --now etcd

# 验证
ETCDCTL_API=3 etcdctl \
  --cacert=/etc/etcd/ssl/ca.pem \
  --cert=/etc/etcd/ssl/etcd.pem \
  --key=/etc/etcd/ssl/etcd-key.pem \
  --endpoints=https://10.0.0.11:2379,https://10.0.0.12:2379,https://10.0.0.13:2379 \
  endpoint health

# 期望：3 个 healthy
```

## Phase 4 · 部署 HAProxy + Keepalived（2 台独立 LB 节点）

### 4.1 安装（lb-01 和 lb-02）

```bash
sudo apt-get install -y haproxy keepalived
```

### 4.2 配置 HAProxy（两台相同）

```bash
cat <<EOF | sudo tee /etc/haproxy/haproxy.cfg
global
    log /dev/log local0
    maxconn 4096
    daemon

defaults
    log     global
    mode    tcp
    option  tcplog
    timeout connect 5s
    timeout client  30s
    timeout server  30s
    retries 3

frontend k8s-api
    bind *:6443
    default_backend k8s-api-backend

backend k8s-api-backend
    option httpchk GET /healthz
    http-check expect status 200
    balance roundrobin
    server master-01 10.0.0.21:6443 check check-ssl verify none
    server master-02 10.0.0.22:6443 check check-ssl verify none
    server master-03 10.0.0.23:6443 check check-ssl verify none

listen stats
    bind *:9090
    mode http
    stats enable
    stats uri /stats
    stats auth admin:admin123
EOF

sudo systemctl enable --now haproxy
```

### 4.3 配置 Keepalived

**lb-01（MASTER）：**

```bash
cat <<EOF | sudo tee /etc/keepalived/keepalived.conf
vrrp_script check_haproxy {
    script "/usr/bin/killall -0 haproxy"
    interval 2
    weight -20
    fall 3
    rise 2
}

vrrp_instance VI_K8S {
    state MASTER
    interface eth0                # 替换为实际网卡名
    virtual_router_id 51
    priority 100
    advert_int 1
    authentication {
        auth_type PASS
        auth_pass K8sHA@2026
    }
    virtual_ipaddress {
        10.0.0.100/24
    }
    track_script {
        check_haproxy
    }
}
EOF
```

**lb-02（BACKUP）：** 同上，仅修改两处：

```
    state BACKUP
    priority 90
```

### 4.4 启动并验证

```bash
sudo systemctl enable --now keepalived

# lb-01 上检查 VIP
ip a | grep 10.0.0.100

# 从任意节点测试
ping -c 3 10.0.0.100
```

### 4.5 故障切换测试

```bash
# 在 lb-01 上停止 HAProxy
sudo systemctl stop haproxy
# 在 lb-02 上检查 VIP 是否漂移过来
ssh lb-02 "ip a | grep 10.0.0.100"    # 应看到 VIP

# 恢复 lb-01
sudo systemctl start haproxy
```

## Phase 5 · 安装 containerd（6 台：3 Master + 3 Worker）

> LB 和 etcd 节点不需要 containerd。

```bash
sudo apt-get update
sudo apt-get install -y ca-certificates curl gnupg
sudo install -m 0755 -d /etc/apt/keyrings
curl -fsSL https://download.docker.com/linux/ubuntu/gpg | sudo gpg --dearmor -o /etc/apt/keyrings/docker.gpg
echo "deb [arch=$(dpkg --print-architecture) signed-by=/etc/apt/keyrings/docker.gpg] https://download.docker.com/linux/ubuntu $(lsb_release -cs) stable" | sudo tee /etc/apt/sources.list.d/docker.list
sudo apt-get update
sudo apt-get install -y containerd.io

sudo mkdir -p /etc/containerd
containerd config default | sudo tee /etc/containerd/config.toml
sudo sed -i 's/SystemdCgroup = false/SystemdCgroup = true/' /etc/containerd/config.toml
sudo systemctl restart containerd
sudo systemctl enable containerd
```

## Phase 7 · 初始化第一个 Master（k8s-master-01）

### 7.1 创建 kubeadm 配置

```bash
cat <<EOF > kubeadm-config.yaml
apiVersion: kubeadm.k8s.io/v1beta4
kind: ClusterConfiguration
kubernetesVersion: "v1.34.0"
controlPlaneEndpoint: "10.0.0.100:6443"
networking:
  podSubnet: "192.168.0.0/16"
  serviceSubnet: "10.96.0.0/12"
etcd:
  external:
    endpoints:
      - https://10.0.0.11:2379
      - https://10.0.0.12:2379
      - https://10.0.0.13:2379
    caFile: /etc/kubernetes/pki/etcd/ca.crt
    certFile: /etc/kubernetes/pki/apiserver-etcd-client.crt
    keyFile: /etc/kubernetes/pki/apiserver-etcd-client.key
---
apiVersion: kubeadm.k8s.io/v1beta4
kind: InitConfiguration
localAPIEndpoint:
  advertiseAddress: "10.0.0.21"
  bindPort: 6443
EOF
```

### 7.2 初始化

```bash
sudo kubeadm init --config=kubeadm-config.yaml --upload-certs
```

### 7.3 配置 kubectl + 保存 join 命令

```bash
mkdir -p $HOME/.kube
sudo cp -i /etc/kubernetes/admin.conf $HOME/.kube/config
sudo chown $(id -u):$(id -g) $HOME/.kube/config
```

保存输出的两条 join 命令（Master 用 / Worker 用）。

## 运维要点

### etcd 自动备份

```bash
# /etc/cron.d/etcd-backup
0 * * * * root ETCDCTL_API=3 /usr/local/bin/etcdctl snapshot save /backup/etcd-$(date +\%Y\%m\%d-\%H).db \
  --cacert=/etc/etcd/ssl/ca.pem --cert=/etc/etcd/ssl/etcd.pem --key=/etc/etcd/ssl/etcd-key.pem \
  --endpoints=https://127.0.0.1:2379 2>&1 | logger -t etcd-backup

# 保留 7 天
0 2 * * * root find /backup -name "etcd-*.db" -mtime +7 -delete
```

### 证书续期

```bash
kubeadm certs check-expiration          # 检查到期时间
sudo kubeadm certs renew all            # 续期所有证书
# etcd 证书有效期 10 年（87600h）
```

浏览器访问 `http://lb-01:9090/stats`（admin/admin123），可查看 3 台 Master 后端健康状态。
